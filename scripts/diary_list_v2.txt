<template>
  <DefaultLayout>
    <div class="diary-page">
      <div class="toolbar glass-sm" style="padding:14px 18px;margin-bottom:20px;display:flex;gap:12px;align-items:center;flex-wrap:wrap;justify-content:space-between">
        <div class="tabs">
          <button :class="['tab', {active: tab === 'all'}]" @click="tab='all';fetch()">&#128293; All</button>
          <button :class="['tab', {active: tab === 'mine'}]" @click="tab='mine';fetch()">&#128214; Mine</button>
        </div>
        <div style="display:flex;gap:8px;align-items:center">
          <el-input v-model="keyword" placeholder="Search..." clearable class="search-inp" @keyup.enter="search"><template #prefix><span style="color:rgba(255,255,255,0.4)">&#128269;</span></template></el-input>
          <el-button type="primary" @click="$router.push('/diaries/new')">+ Write</el-button>
        </div>
      </div>
      <div v-if="loading" class="empty">Loading...</div>
      <div v-else-if="!diaries.length" class="empty">No diaries yet</div>
      <div v-else class="grid">
        <div v-for="d in diaries" :key="d.id" class="card glass" @click="$router.push('/diaries/' + d.id)">
          <div v-if="d.images?.length" class="card-img"><img :src="d.images[0]" /></div>
          <div v-else class="card-img-placeholder">&#128221;</div>
          <div class="card-body">
            <div class="card-badges">
              <span v-if="d.images?.length" class="badge badge-cyan">&#128247; {{ d.images.length }}</span>
              <span v-if="d.videoMeta" class="badge badge-yellow">&#127916;</span>
              <span v-if="d.musicUrl" class="badge badge-green">&#127925;</span>
            </div>
            <h3>{{ d.title }}</h3>
            <p class="card-desc">{{ (d.content || '').substring(0, 100) }}{{ (d.content || '').length > 100 ? '...' : '' }}</p>
            <div class="card-foot">
              <span>&#11088; {{ d.avgRating?.toFixed(1) || '—' }}</span>
              <span>&#128065; {{ d.popularity }}</span>
              <span>{{ d.createdAt?.substring(0,10) || '' }}</span>
            </div>
          </div>
        </div>
      </div>
    </div>
  </DefaultLayout>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import DefaultLayout from '@/layouts/DefaultLayout.vue'
import { diaryApi } from '@/api/diaryApi'
import type { DiaryResponse } from '@/types/api'

const diaries = ref<DiaryResponse[]>([])
const loading = ref(false)
const keyword = ref('')
const tab = ref('all')

async function fetch() {
  loading.value = true
  try {
    const r = tab.value === 'mine' ? await diaryApi.mine({ size: 30 }) : await diaryApi.list({ size: 30 })
    diaries.value = r.data.data.content || []
  } catch {} finally { loading.value = false }
}
function search() {
  if (!keyword.value.trim()) { fetch(); return }
  loading.value = true
  diaryApi.search(keyword.value).then(r => { diaries.value = r.data.data.content || [] }).finally(() => loading.value = false)
}
onMounted(fetch)
</script>
